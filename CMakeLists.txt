cmake_minimum_required(VERSION 4.2.3)

project(toaster
    VERSION 0.2.0
    DESCRIPTION "Windows toast notifications and .wav playback from the command line"
    LANGUAGES CXX RC)

if(NOT MSVC)
    message(FATAL_ERROR "toaster targets Windows and must be built with MSVC.")
endif()

# Statically link the C/C++ runtime so the result is a single self-contained .exe.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
include(EmbedAssets)

# Every .wav/.mp3 in TOASTER_SOUNDS_DIR and every image in TOASTER_IMAGES_DIR is
# compiled into the .exe. Relative paths are resolved against the source
# directory. Adding or removing files re-runs CMake automatically.
#   cmake --preset msvc-x64 -DTOASTER_SOUNDS_DIR=C:/my/sounds -DTOASTER_IMAGES_DIR=C:/my/images
set(TOASTER_SOUNDS_DIR "assets/sounds" CACHE PATH "Folder whose .wav/.mp3 files are embedded in toaster.exe")
set(TOASTER_IMAGES_DIR "assets/images" CACHE PATH "Folder whose .png/.jpg/.jpeg/.gif files are embedded in toaster.exe")

toaster_collect_assets(TOASTER_EMBED_SOUNDS "${TOASTER_SOUNDS_DIR}" wav mp3)
toaster_collect_assets(TOASTER_EMBED_IMAGES "${TOASTER_IMAGES_DIR}" png jpg jpeg gif)

# Sound packs: *.json files in TOASTER_SOUNDS_DIR describing categories of
# sounds (see cmake/SoundPacks.cmake). Their sounds and icons are embedded too.
include(SoundPacks)
toaster_load_sound_packs("${TOASTER_SOUNDS_DIR}" TOASTER_PACK_SOUNDS TOASTER_PACK_IMAGES
                         "${CMAKE_CURRENT_BINARY_DIR}/generated/sound_packs.h"
                         "${TOASTER_IMAGES_DIR}")
# Don't embed a file twice when it is both loose in the folder and in a pack.
foreach(item IN LISTS TOASTER_PACK_SOUNDS)
    string(REGEX REPLACE "^[^|]*\\|" "" pack_file "${item}")
    list(REMOVE_ITEM TOASTER_EMBED_SOUNDS "${pack_file}")
endforeach()
list(APPEND TOASTER_EMBED_SOUNDS ${TOASTER_PACK_SOUNDS})
foreach(item IN LISTS TOASTER_PACK_IMAGES)
    string(REGEX REPLACE "^[^|]*\\|" "" pack_file "${item}")
    list(REMOVE_ITEM TOASTER_EMBED_IMAGES "${pack_file}")
endforeach()
list(APPEND TOASTER_EMBED_IMAGES ${TOASTER_PACK_IMAGES})

configure_file(src/version.h.in ${CMAKE_CURRENT_BINARY_DIR}/generated/version.h @ONLY)

add_executable(toaster
    src/main.cpp
    src/cli.cpp
    src/console.cpp
    src/embedded.cpp
    src/app_id.cpp
    src/toast.cpp
    src/sound.cpp
    src/packs.cpp
    src/mp3.cpp
    src/skill.cpp
)

toaster_embed_assets(toaster
    SOUNDS ${TOASTER_EMBED_SOUNDS}
    IMAGES ${TOASTER_EMBED_IMAGES}
)

# The AI agent skill that `toaster --install-skill` writes out is compiled in too.
set(TOASTER_SKILL_FILE "${CMAKE_CURRENT_SOURCE_DIR}/skill/toaster/SKILL.md")
set(TOASTER_SKILL_RC "${CMAKE_CURRENT_BINARY_DIR}/generated/skill.rc")
file(TO_CMAKE_PATH "${TOASTER_SKILL_FILE}" toaster_skill_rc_path)
toaster_write_if_changed("${TOASTER_SKILL_RC}"
    "// Generated by CMakeLists.txt. Do not edit.\nSKILL_TOASTER RCDATA \"${toaster_skill_rc_path}\"\n")
target_sources(toaster PRIVATE "${TOASTER_SKILL_RC}")
set_source_files_properties("${TOASTER_SKILL_RC}" PROPERTIES OBJECT_DEPENDS "${TOASTER_SKILL_FILE}")

target_include_directories(toaster PRIVATE
    src
    ${CMAKE_CURRENT_BINARY_DIR}/generated
    third_party/dr_libs   # dr_mp3.h: header-only MP3 decoder (public domain / MIT-0)
)

target_compile_features(toaster PRIVATE cxx_std_20)
set_target_properties(toaster PROPERTIES CXX_EXTENSIONS OFF)

target_precompile_headers(toaster PRIVATE src/pch.h)

target_compile_definitions(toaster PRIVATE
    UNICODE
    _UNICODE
    WIN32_LEAN_AND_MEAN
    NOMINMAX
    _WIN32_WINNT=0x0A00   # Windows 10 / 11
)

target_compile_options(toaster PRIVATE
    /W4
    /permissive-
    /utf-8
    /EHsc
    /Zc:__cplusplus
    /MP
    /bigobj
    $<$<NOT:$<CONFIG:Debug>>:/O2 /GL /Gy>
)

# Only Windows system import libraries; nothing extra ships next to the .exe.
target_link_libraries(toaster PRIVATE
    WindowsApp   # C++/WinRT / Windows Runtime
    winmm        # PlaySound
    shlwapi      # UrlCreateFromPath
)

target_link_options(toaster PRIVATE
    $<$<NOT:$<CONFIG:Debug>>:/LTCG /OPT:REF /OPT:ICF>
)

# Tests: `ctest --preset release` after building. See tests/CMakeLists.txt.
option(TOASTER_BUILD_TESTS "Build the test suite" ON)
option(TOASTER_TEST_UI "Register the tests that show real toasts and play audio" OFF)
if(TOASTER_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
